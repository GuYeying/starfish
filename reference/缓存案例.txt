use std::fs;
use wgpu::*;

// ------------------------------
// 1. 缓存保存/加载工具
// ------------------------------
fn load_pipeline_cache(device: &Device, adapter: &Adapter) -> PipelineCache {
    // 按显卡生成唯一缓存文件名（不同显卡不能混用缓存）
    let cache_key = util::pipeline_cache_key(adapter);
    let path = format!("pipeline_cache_{}.bin", cache_key);

    // 尝试读取上次缓存
    let data = fs::read(path).ok();

    // 创建缓存对象
    device.create_pipeline_cache(&PipelineCacheDescriptor {
        label: Some("Pipeline Cache"),
        data,
    })
}

fn save_pipeline_cache(cache: &PipelineCache, adapter: &Adapter) {
    // 只有 Vulkan 后端才有可保存的数据
    if let Some(data) = cache.get_data() {
        let cache_key = util::pipeline_cache_key(adapter);
        let path = format!("pipeline_cache_{}.bin", cache_key);
        let _ = fs::write(path, data);
    }
}

// ------------------------------
// 2. 创建带缓存的管线
// ------------------------------
fn create_pipeline_with_cache(
    device: &Device,
    cache: &PipelineCache,
    vs_module: &ShaderModule,
    fs_module: &ShaderModule,
    pipeline_layout: &PipelineLayout,
) -> RenderPipeline {
    device.create_render_pipeline(&RenderPipelineDescriptor {
        label: Some("Test Pipeline"),
        // ✅ 关键：把缓存绑进去
        // Vulkan：会自动用缓存、跳过编译
        // OpenGL/DX12/Metal：自动忽略，不报错
        cache: Some(cache),
        layout: Some(pipeline_layout),
        vertex: VertexState {
            module: vs_module,
            entry_point: "vs_main",
            buffers: &[],
        },
        fragment: Some(FragmentState {
            module: fs_module,
            entry_point: "fs_main",
            targets: &[Some(ColorTargetState {
                format: TextureFormat::Bgra8UnormSrgb,
                blend: None,
                write_mask: ColorWrites::ALL,
            })],
        }),
        primitive: PrimitiveState::default(),
        depth_stencil: None,
        multisample: MultisampleState::default(),
        multiview: None,
    })
}

// ------------------------------
// 3. 主流程（最简演示）
// ------------------------------
fn main() {
    // 1. 初始化实例、适配器、设备
    let instance = Instance::default();
    let adapter = pollster::block_on(wgpu::util::initialize_adapter(&instance)).unwrap();
    let (device, _queue) = pollster::block_on(adapter.request_device(&Default::default(), None)).unwrap();

    // 2. 加载缓存（如果有上次的缓存）
    let pipeline_cache = load_pipeline_cache(&device, &adapter);

    // 3. 创建着色器（必须传，缓存不会省掉这个）
    let vs_module = device.create_shader_module(ShaderModuleDescriptor {
        label: Some("VS"),
        source: ShaderSource::Wgsl(
            "
            @vertex
            fn vs_main(@builtin(vertex_index) i: u32) -> @builtin(position) vec4<f32> {
                let pos = array(vec2(-1,-1), vec2(3,-1), vec2(-1,3));
                return vec4(pos[i], 0, 1);
            }
            "
            .into(),
        ),
    });
    let fs_module = device.create_shader_module(ShaderModuleDescriptor {
        label: Some("FS"),
        source: ShaderSource::Wgsl(
            "
            @fragment
            fn fs_main() -> @location(0) vec4<f32> {
                return vec4(1,0.5,0,1);
            }
            "
            .into(),
        ),
    });

    // 4. 创建管线布局
    let pipeline_layout = device.create_pipeline_layout(&PipelineLayoutDescriptor {
        label: None,
        bind_group_layouts: &[],
        push_constant_ranges: &[],
    });

    // 5. 创建管线（自动使用缓存）
    let pipeline = create_pipeline_with_cache(
        &device,
        &pipeline_cache,
        &vs_module,
        &fs_module,
        &pipeline_layout,
    );

    // 6. 退出时保存缓存（只有 Vulkan 会真正写入文件）
    save_pipeline_cache(&pipeline_cache, &adapter);

    println!("管线创建完成！缓存已自动加载/保存");
}
